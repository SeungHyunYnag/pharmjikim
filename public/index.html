async function loadPharmacies(lat,lng){ const res=await fetch(`/api/pharmacies?lat=${lat}&lng=${lng}`); const data=await res.json(); let hasNearbyOpen=false; data.forEach(p=>{ const marker=new naver.maps.Marker({ position:new naver.maps.LatLng(p.lat,p.lng), map:map }); const dist=getDistance(lat,lng,p.lat,p.lng); const open=isOpen(p.start,p.end); if(dist<=1&&open) hasNearbyOpen=true; // 🔥 좌표 기반 + placePath 방식 (핵심 수정) const naverLink = `https://map.naver.com/v5/?c=${p.lng},${p.lat},16,0,0,0,dh&placePath=${encodeURIComponent(p.name)}`; const infoWindow=new naver.maps.InfoWindow({ maxWidth:250, content:`
${p.name}

📞 ${p.tel}
🕒 ${formatTime(p.start)} ~ ${formatTime(p.end)} ${open?"🟢 운영중":"🔴 종료"}
📍 ${p.addr || ""}
※ 보다 정확한 운영 여부는 전화로 확인 바랍니다
🔗 휴일지킴이 약국
🔗 네이버 지도에서 보기
` }); naver.maps.Event.addListener(marker,"click",()=>{ hideNotice(); // 기존 열린 창 닫기 if(currentInfoWindow){ currentInfoWindow.close(); } infoWindow.open(map,marker); currentInfoWindow=infoWindow; }); }); if(!hasNearbyOpen){ showNotice(); } }
